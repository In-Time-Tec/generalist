import { layer } from "@effect/platform-bun/BunHttpServer"
import { runMain } from "@effect/platform-bun/BunRuntime"
import { Layer } from "effect"
import { FetchHttpClient, HttpRouter } from "effect/unstable/http"
import { appLayer, sessionRegistryLayer } from "./server"

const serverLayer = HttpRouter.serve(appLayer).pipe(
  Layer.provideMerge(layer({ port: 4000 })),
  Layer.provideMerge(sessionRegistryLayer),
  Layer.provideMerge(FetchHttpClient.layer),
)

runMain(Layer.launch(serverLayer))
